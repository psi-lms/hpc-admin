# AiiDA backups on thanos. One line per project; profiles are passed as a
# space-separated list so a project is backed up by a SINGLE process.
#
# Why one line per project: aiida-backup.sh derives its log path from the
# project name ($ROOT/<project>/backup.log) and opens it with a truncating
# `tee`. Two cron entries for the same project race on that file. Passing
# multiple profiles to one invocation makes the script loop over them
# sequentially into one log instead.
#
# Start times are staggered, largest project first, so ~200 GB of pg_dump
# and the NFS writes are not all issued at 00:00.

REPO=/home/aiida/scripts/aiida-backup-script
aiida_venv_dir=/home/aiida/.aiida_venvs
